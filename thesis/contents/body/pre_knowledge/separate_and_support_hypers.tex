\section{Siêu phẳng phân chia và siêu phẳng hỗ trợ}
\label{section:separating_and_supporting_hyperplanes}

\subsection{Định lí siêu phẳng phân chia}
\begin{theorem}
    Giả sử $\mathcal{C}$ và $\mathcal{D}$ là hai tập lồi khác rỗng và rời nhau, tức là $\mc{C} \cap \mc{D} = \emptyset$. Khi đó tồn tại $\mb{a} \neq 0$ và $b$ sao cho  
    \[
    \mb{a}^T \mb{x} \leq b, \quad \forall \mb{x} \in \mc{C}
    \]
    và  
    \[
    \mb{a}^T \mb{x} \geq b, \quad \forall \mb{x} \in \mc{D}.
    \]
    Nói cách khác, hàm affine $\mb{a}^T \mb{x} - b$ là không dương trên $\mc{C}$ và không âm trên $\mc{D}$.  
    Siêu phẳng $\{\mb{x} \mid \mb{a}^T \mb{x} = b\}$ được gọi là \textbf{siêu phẳng phân chia} hai tập $\mc{C}$ và $\mc{D}$.
\end{theorem}
